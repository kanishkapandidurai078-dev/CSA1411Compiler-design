%{
#include <stdio.h>
int pos = 0, neg = 0;
%}
%%

-[0-9]+ {neg++;}

[0-9]+ {pos++;}

[ \t\n]+ ;

. ;

%%

int yywrap(void) { return 1; }

int main()
{
    printf("Enter numbers:\n");
    yylex();

    printf("\nPositive Numbers = %d", pos);
    printf("\nNegative Numbers = %d\n", neg);

    return 0;
}
